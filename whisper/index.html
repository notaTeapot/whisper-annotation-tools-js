Whisper Annotation Converter

Bulk-Verarbeitung: Paart Whisper-JSON mit Metadaten und strukturiert sie exakt nach structure.py

Verarbeitungs-Logik (Dynamisches Beispiel)

Input-Paar (Automatisch verknüpft)

// 1. file_metadata.json
{
  "name": "Interview 1",
  "date": "2024-05-12",
  "spotify_url": "https://..."
}

// 2. file.json (Whisper)
[
  { "text": "Und ", "timestamp": [0.0, 1.0], "speaker": "SPK_0" },
  { "text": "dann.", "timestamp": [1.0, 1.5], "speaker": "SPK_0" }
]
➔

Output: .flk (Folker Format)

<folker-transcription>
  <head>
    <transcription-log>
      <log-entry start="2024/..." end="2024/..." who="system">Generated by whisper-annotation-tools Interview 1, 2024-05-12, https://...</log-entry>
    </transcription-log>
  </head>
  <speakers>
    <speaker speaker-id="SPK_0">
      <name>SPK_0</name>
    </speaker>
  </speakers>
  <recording path="file.wav" />
  <timeline>
    <timepoint timepoint-id="TLI_1" absolute-time="0" />
    <timepoint timepoint-id="TLI_2" absolute-time="1.5" />
  </timeline>
  <contribution speaker-reference="SPK_0" start-reference="TLI_1" end-reference="TLI_2">
    <unparsed>Und dann.</unparsed>
  </contribution>
</folker-transcription>

Output: XML (Sketch Engine)

<!-- Generated by whisper-annotation-tools -->
<doc name="Interview 1" date="2024-05-12" spotify_url="https://...">
  <p speaker="SPK_0" timestamp="[0, 1.5]" timestamp_h="['00:00:00', '00:00:01']">
    <s timestamp="[0, 1.5]" timestamp_h="['00:00:00', '00:00:01']">Und dann.</s>
  </p>
</doc>
